#include #include #include #include #include #include // Nécessaire pour lire la MAC address // --------------------------------------------------------- // PARAMÈTRES BLUETOOTH // --------------------------------------------------------- // Les UUIDs doivent correspondre EXACTEMENT à ceux du code JavaScript (Web Bluetooth) #define SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b" #define CHARACTERISTIC_UUID "beb5483e-36e1-4688-b7f5-ea07361b26a8" BLEServer* pServer = NULL; BLECharacteristic* pCharacteristic = NULL; bool deviceConnected = false; bool oldDeviceConnected = false; // Tableaux pour stocker les noms générés dynamiquement char deviceName[25]; char droneId[25]; // --------------------------------------------------------- // VARIABLES DE SIMULATION // --------------------------------------------------------- float currentLat = 48.8566; // Point de départ arbitraire float currentLon = 2.3522; int currentAlt = 0; int currentRssi = -40; // --------------------------------------------------------- // CALLBACKS BLUETOOTH (Gestion de la connexion) // --------------------------------------------------------- class MyServerCallbacks: public BLEServerCallbacks { void onConnect(BLEServer* pServer) { deviceConnected = true; Serial.println("Client BLE connecté !"); }; void onDisconnect(BLEServer* pServer) { deviceConnected = false; Serial.println("Client BLE déconnecté."); } }; // --------------------------------------------------------- // INITIALISATION // --------------------------------------------------------- void setup() { Serial.begin(115200); Serial.println("--- Démarrage du Serveur DLXR_RX ---"); // 1. Récupération de l'adresse MAC physique unique de l'ESP32 uint8_t mac[6]; esp_efuse_mac_get_default(mac); // 2. Création des noms en utilisant les 2 derniers octets de la MAC // Le préfixe "DLXR_RX_" est requis par le filtre JS (namePrefix) sprintf(deviceName, "DLXR_RX_%02X%02X", mac[4], mac[5]); sprintf(droneId, "Drone_%02X%02X", mac[4], mac[5]); Serial.print("Nom Bluetooth généré : "); Serial.println(deviceName); Serial.print("ID Drone généré : "); Serial.println(droneId); // 3. Initialisation du périphérique BLE avec le nom dynamique BLEDevice::init(deviceName); // 4. Création du serveur et assignation des Callbacks pServer = BLEDevice::createServer(); pServer->setCallbacks(new MyServerCallbacks()); // 5. Création du Service BLEService *pService = pServer->createService(SERVICE_UUID); // 6. Création de la Caractéristique (Lecture + Notifications) pCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_NOTIFY ); // Indispensable pour que l'interface Web (GATT) puisse s'abonner aux notifications pCharacteristic->addDescriptor(new BLE2902()); // 7. Démarrage du service et de l'annonce (Advertising) pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->setScanResponse(false); pAdvertising->setMinPreferred(0x0); // Aide à la découverte rapide (ex: iOS) BLEDevice::startAdvertising(); Serial.println("En attente de connexion client via l'interface Web..."); } // --------------------------------------------------------- // BOUCLE PRINCIPALE // --------------------------------------------------------- void loop() { // Si le téléphone/navigateur est connecté, on génère et envoie la trame if (deviceConnected) { // --- SIMULATION DES MOUVEMENTS --- // À remplacer par la lecture de tes vraies données de télémétrie (ex: liaison série vers récepteur) currentLat += 0.0001; currentLon += 0.00005; currentAlt = (currentAlt < 120) ? currentAlt + 1 : 120; currentRssi = random(-90, -40); String etat = (currentAlt < 5) ? "GROUND" : "HIGH"; // --- CRÉATION DU JSON (avec ArduinoJson 7) --- JsonDocument doc; doc["drone_id"] = droneId; // Utilisation de l'ID généré dynamiquement doc["etat"] = etat; doc["alt"] = currentAlt; doc["dist"] = currentAlt * 2.5; doc["rssi"] = currentRssi; doc["lat"] = currentLat; doc["lon"] = currentLon; // Sérialisation du JSON dans une chaîne String jsonString; serializeJson(doc, jsonString); // --- ENVOI VIA BLUETOOTH --- pCharacteristic->setValue(jsonString.c_str()); pCharacteristic->notify(); Serial.print("Télémétrie envoyée : "); Serial.println(jsonString); // Fréquence d'envoi : 500ms (2 Hz) delay(500); } // --- GESTION DES DÉCONNEXIONS --- // Si on vient de perdre la connexion, on relance l'Advertising pour permettre une reconnexion if (!deviceConnected && oldDeviceConnected) { delay(500); // Laisse le temps à la pile Bluetooth de se nettoyer pServer->startAdvertising(); Serial.println("Redémarrage de l'Advertising..."); oldDeviceConnected = deviceConnected; } // Si on vient de se connecter, on met à jour l'état if (deviceConnected && !oldDeviceConnected) { oldDeviceConnected = deviceConnected; } }
